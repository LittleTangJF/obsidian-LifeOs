**数组扁平化**

将嵌套数组 `[1, [2, [3, 4], 5]]` 扁平化为 `[1, 2, 3, 4, 5]`

```js
function flatten(arr) {
    let result = [];
    for (let item of arr) {
        if (Array.isArray(item)) {
            result = result.concat(flatten(item)); // 递归扁平化
        } else {
            result.push(item);
        }
    }
    return result;
}

console.log(flatten([1, [2, [3, 4], 5]])); // 输出 [1, 2, 3, 4, 5]
```
