**考察内容**

- 熟悉 `data-*` 的使用，用来存储额外信息（以下的data-id）
- `e` 是 React 封装的合成事件，可以从 `e.currentTarget` 取当前绑定事件的 DOM。
- TS类型是明确事件类型是 `React.MouseEvent<HTMLButtonElement>`


```ts
import React, { useState } from 'react';

const ListComponent: React.FC = () => {
  const list = ['a', 'b', 'c'];
  const [active, setActive] = useState<string | null>(null);

  const handleClick = (e: React.MouseEvent<HTMLButtonElement>) => {
    // 从自定义属性中取出 id
    const id = (e.currentTarget.dataset.id as string) || '';
    setActive(id);
  };

  return (
    <div>
      {list.map((item) => (
        <button
          key={item}
          data-id={item}
          onClick={handleClick}
          style={{
            marginRight: 8,
            background: active === item ? '#4CAF50' : '#eee',
          }}
        >
          {item}
        </button>
      ))}
      {active && <p>当前选中：{active}</p>}
    </div>
  );
};

export default ListComponent;

```
